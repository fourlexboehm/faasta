13const CONFIG_FILE: &str = "faasta/github_auth.json";
14const USER_AGENT: &str = "faasta-cli";
15const MACHINE_MAN_PREVIEW: &str = "application/vnd.github.machine-man-preview+json";
16/// Installation access tokens expire after an hour; refresh a little early.
17const TOKEN_REFRESH_INTERVAL: Duration = Duration::from_secs(55 * 60);

22    exp: u64,
23    iss: u64,
24}

39/// GitHub app installation token, refreshed automatically before it expires.
40struct InstallationAccessToken {
41    token: String,
42    refresh_at: Instant,
43    params: GithubAuthParams,
44}
45
46impl InstallationAccessToken {
47    async fn new(params: GithubAuthParams) -> Result<Self, Error> {
48        let raw = fetch_installation_token(&params).await?;
49        Ok(Self {
50            token: raw.token,
51            refresh_at: Instant::now() + TOKEN_REFRESH_INTERVAL,
52            params,
53        })
54    }
55
56    async fn header(&mut self) -> Result<HeaderMap, Error> {
57        if Instant::now() >= self.refresh_at {
58            let raw = fetch_installation_token(&self.params).await?;
59            self.token = raw.token;
60            self.refresh_at = Instant::now() + TOKEN_REFRESH_INTERVAL;
61        }
62
63        let mut headers = HeaderMap::new();
64        headers.insert(
65            http::header::AUTHORIZATION,
66            format!("token {}", self.token).parse()?,
67        );
68        Ok(headers)
69    }
70}
71
72async fn fetch_installation_token(
73    params: &GithubAuthParams,
74) -> Result<RawInstallationAccessToken, Error> {
75    let now = SystemTime::now().duration_since(UNIX_EPOCH)?.as_secs();
76    let claims = JwtClaims {
77        iat: now,
78        exp: now + 60,
79        iss: params.app_id,
80    };
81    let header = Header {
82        alg: Algorithm::RS256,
83        ..Default::default()
84    };
85    let key = EncodingKey::from_rsa_pem(&params.private_key)?;
86    let jwt = jsonwebtoken::encode(&header, &claims, &key)?;
87
88    let url = format!(
89        "https://api.github.com/app/installations/{}/access_tokens",
90        params.installation_id
91    );
92    let response = HttpClient::new()?
93        .post(&url)?
94        .bearer_auth(jwt)?
95        .header("Accept", MACHINE_MAN_PREVIEW)?
96        .header("User-Agent", params.user_agent.as_str())?
97        .send()
98        .await?;
99
100    let status = response.status();
101    if !status.is_success() {
102        let body = response.text().await.unwrap_or_default();
103        anyhow::bail!("GitHub installation token request failed ({status}): {body}");
104    }
105
106    Ok(response.json().await?)
107}

118/// Manages GitHub authentication for the CLI
119pub struct GitHubAuth {
120    config: AuthConfig,
121    token: Option<InstallationAccessToken>,
122    config_path: PathBuf,
123}
124
125impl GitHubAuth {
126    /// Initialize GitHub authentication
127    pub async fn new() -> Result<Self, Error> {
128        let config_path = Self::get_config_path()?;
129        let config = Self::load_config(&config_path).await?;
130
131        Ok(Self {
132            config,
133            token: None,
134            config_path,
135        })
136    }
137
138    /// Authenticate with GitHub
139    pub async fn authenticate(&mut self) -> Result<(), Error> {
140        if self.config.app_id == 0
141            || self.config.installation_id == 0
142            || self.config.private_key.is_empty()
143        {
144            return Err(anyhow::anyhow!(
145                "GitHub App not configured. Run 'cargo faasta auth setup' first."
146            ));
147        }
148
149        self.token = Some(
150            InstallationAccessToken::new(GithubAuthParams {
151                user_agent: USER_AGENT.into(),
152                private_key: self.config.private_key.clone(),
153                app_id: self.config.app_id,
154                installation_id: self.config.installation_id,
155            })
156            .await?,
157        );
158
159        // Get user ID if we don't have it yet
160        if self.config.user_id.is_none() {
161            self.fetch_and_store_user_id().await?;
162        }
163
164        Ok(())
165    }
166
167    /// Get authentication header for API requests
168    pub async fn header(&mut self) -> Result<HeaderMap, Error> {
169        if self.token.is_none() {
170            self.authenticate().await?;
171        }
172
173        self.token.as_mut().unwrap().header().await
174    }
175
176    /// Store project HMAC for ownership verification
177    pub async fn store_project_hmac(
178        &mut self,
179        project_name: &str,
180        hmac: &str,
181    ) -> Result<(), Error> {
182        self.config
183            .project_hmacs
184            .insert(project_name.to_string(), hmac.to_string());
185        self.save_config().await?;
186        Ok(())
187    }
188
189    /// Get project HMAC if it exists
190    pub fn get_project_hmac(&self, project_name: &str) -> Option<&String> {
191        self.config.project_hmacs.get(project_name)
192    }
193
194    /// Check if a project is owned by the current user
195    pub fn owns_project(&self, project_name: &str) -> bool {
196        self.config.project_hmacs.contains_key(project_name)
197    }
198
199    /// Get the list of projects owned by the user
200    pub fn get_owned_projects(&self) -> Vec<String> {
201        self.config.project_hmacs.keys().cloned().collect()
202    }
203
204    /// Check if user has reached project limit
205    pub fn has_reached_project_limit(&self) -> bool {
206        self.config.project_hmacs.len() >= 5
207    }
208
209    /// Setup GitHub app credentials
210    pub async fn setup(
211        &mut self,
212        app_id: u64,
213        installation_id: u64,
214        private_key: Vec<u8>,
215    ) -> Result<(), Error> {
216        self.config.app_id = app_id;
217        self.config.installation_id = installation_id;
218        self.config.private_key = private_key;
219        self.save_config().await?;
220        Ok(())
221    }
222
223    /// Get user ID from authenticated GitHub instance
224    async fn fetch_and_store_user_id(&mut self) -> Result<(), Error> {
225        let header = self.token.as_mut().unwrap().header().await?;
226
227        // Create authenticated client
228        let response = HttpClient::new()?
229            .get("https://api.github.com/app")?
230            .headers(header)
231            .send()
232            .await?;
233
234        if response.status().is_success() {
235            let app_info: serde_json::Value = response.json().await?;
236            if let Some(id) = app_info.get("id").and_then(|v| v.as_str()) {
237                self.config.user_id = Some(id.to_string());
238                self.save_config().await?;
239            }
240        }
241
242        Ok(())
243    }
244
245    /// Get the path to the config file
246    fn get_config_path() -> Result<PathBuf, Error> {
247        let mut path =
248            config_dir().ok_or_else(|| anyhow::anyhow!("Could not find user config directory"))?;
249
250        path.push(CONFIG_FILE);
251
252        // Ensure parent directory exists
253        if let Some(parent) = path.parent()
254            && !parent.exists()
255        {
256            fs::create_dir_all(parent)?;
257        }
258
259        Ok(path)
260    }
261
262    /// Load config from disk
263    async fn load_config(path: &Path) -> Result<AuthConfig, Error> {
264        if path.exists() {
265            let data = compio::fs::read(path).await?;
266            let content = String::from_utf8(data)?;
267            Ok(serde_json::from_str(&content)?)
268        } else {
269            // Create default config
270            let default_config = AuthConfig::default();
271            let content = serde_json::to_string_pretty(&default_config)?;
272            let BufResult(result, _) = compio::fs::write(path, content.into_bytes()).await;
273            result?;
274            Ok(default_config)
275        }
276    }
277
278    /// Save config to disk
279    pub async fn save_config(&self) -> Result<(), Error> {
280        let content = serde_json::to_string_pretty(&self.config)?;
281        let BufResult(result, _) = compio::fs::write(&self.config_path, content.into_bytes()).await;
282        result?;
283        Ok(())
284    }
285
286    /// Check if GitHub app is configured
287    pub fn is_configured(&self) -> bool {
288        self.config.app_id != 0
289            && self.config.installation_id != 0
290            && !self.config.private_key.is_empty()
291    }
292
293    /// Get user ID if available
294    pub fn get_user_id(&self) -> Option<&str> {
295        self.config.user_id.as_deref()
296    }
